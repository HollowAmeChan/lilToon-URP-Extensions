P=np.dtype([('position','<f4',(3,)),('misc','<u4'),('radiance','<u4'),('weight','<f4'),('normal','<u4'),('unused','<u4')])
def raw(p):
 if p.exists():return p.read_bytes()
 q=p.with_suffix(p.suffix+'.gz');return gzip.decompress(q.read_bytes()) if q.exists() else lzma.decompress(p.with_suffix(p.suffix+'.xz').read_bytes())
def read(p):return json.loads(raw(p).decode('utf-8-sig'))
def intersect(o,d,low,high,phase):
 n=len(o);best=np.full(n,np.inf);which=np.full(n,-1,dtype=int)
 planes=[(0,0,1,1),(-1.25 if phase==1 else -1,0,2,2),(-.6,-.35 if phase==3 else 10,.2,2),(-.5,0 if phase in [5,6] else 10,2,2)]
 for i,(z,cx,hx,hy) in enumerate(planes):
  if i==3 and phase!=5:continue
  dz=d[:,2];t=np.divide(z-o[:,2],dz,out=np.full(n,np.inf),where=np.abs(dz)>1e-20);hit=o+d*t[:,None]
  ok=(t>=low)&(t<=high)&(np.abs(hit[:,0]-cx)<=hx+1e-6)&(np.abs(hit[:,1])<=hy+1e-6)&(t<best);best[ok]=t[ok];which[ok]=i
 return best,which

def visible(o,p,bias,phase):
 delta=p-o;dist=np.linalg.norm(delta,axis=1);direction=delta/np.maximum(dist[:,None],1e-12);t,which=intersect(o,direction,bias,dist-bias,phase);return (which<0)&(dist>2*bias)

  if fi:

  use=valid&matched

   row['materialMismatches']+=int(np.count_nonzero(np.abs(base-v[use,7,:3])>1e-6));assert np.all(v[use,6,3]==1) and np.all(v[use,7,3]==0) and np.all(v[use,8]==0)
   current=v[use,5,:3].astype(float);currentOpen=visible(primary[use],current,bias[use],phase);row['visibilityMismatches']+=int(np.count_nonzero(currentOpen!=(v[use,9,1]>0)))
   lightIndex=(u[use,0,1]&0x7fffffff).astype(int);has=(u[use,0,1]&0x80000000)!=0;expected=np.zeros((len(idx),3))

  assert np.all(np.isfinite(v[:,1:18]))

 if not all(x['geometryHitMismatches']==x['visibilityMismatches']==x['materialMismatches']==x['oracleRadianceViolations']==0 for x in counts):print('ORACLE FAILURE',counts,'radiance',radianceError)
 assert all(x['geometryHitMismatches']==x['visibilityMismatches']==x['materialMismatches']==x['oracleRadianceViolations']==0 for x in counts)
